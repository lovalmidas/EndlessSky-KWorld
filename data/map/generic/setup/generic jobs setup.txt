mission "KW.Generic.Jobs.__a.Job Limit Rural"
	invisible
	repeat
	landing
	to accept
		has "flagship planet attribute: rural"
		not "flagship planet attribute: urban"
	on accept
		"job-board: limit: cargo jobs [S]" = 3
		"job-board: limit: cargo jobs [M]" = 1
		"job-board: limit: cargo jobs [L]" = 0
		"job-board: limit: passenger jobs [S]" = 6
		"job-board: limit: passenger jobs [M]" = 4
		"job-board: limit: passenger jobs [L]" = 0
		fail
