### Naming ###
#	ID: "KW.Generic.Jobs.<priority><index><cargo type>.<short name>"
#	Example: "KW.Generic.Jobs.90f.Clothing Cargo [0]"
#		Sizes:
#			_ = special
#			0 = special
#			1 = 
#			2 = mega, deadline
#			3 = mega
#			4 = 
#			5 = bulk, deadline
#			6 = bulk
#			7 = 
#			8 = normal, deadline
#			9 = normal
#		Cargo Types:
#			f = food
#			c = clothing
#			m = metal
#			p = plastic
#			q = equipment
#			i = industrial
#			d = medical
#			e = electronics
#			h = heavy metal
#			x = luxury goods
#			n = construction
#			y = military
#			s = ship alloys
#			r = research samples
#
# Game orders missions by internal name, so priority 0 is evaluated first
# Spread priority and index so other cargoes have a (near-equal) chance to appear

### COMMODITY: INDUSTRIAL ###
# Industrial equipment is needed primarily by factories
# Factories typically don't take small orders
#
#		[S] ---
#		[M] hitech / logistics -> factory / large
#		[L] hitech / logistics -> factory + large
#		exclude `isolated`; they take no generic jobs
#		exclude `logistics` destinations; logistics-to-logistics have their own jobs
#
# Use these amounts
#		Normal:	5 2 .1
#		Bulk:	20 2 .075
#		Mega:	30 3 .04
#
# Use these distances
#		[S]	---
#		[M]	1-2, 3-5, 6-8
#		[L]	1-7, 8-12
#		
# Payment
#		[S]	0 + 165 * d
#		[M]	0 + 170 * d
#		[L]	0 + 175 * d
#
# 
mission "KW.Generic.Jobs.60i.Industrial Bulk Cargo [0]"
	name "📦Bulk Delivery of Industrial Equipment to <planet>"
	job
	repeat
	description "${KW.Phrase.Job: cargo offer}"
	"offer precedence" 100
	color selected "KW.Color: job cargo selected"
	color unselected "KW.Color: job cargo unselected"
	color unavailable "KW.Color: job cargo unavailable"
	cargo "Industrial" 20 2 .075
	"distance calculation settings"
		"all wormholes"
	to offer
		has "job-board: enable generic cargo jobs"
		"job-board: cargo jobs [M]" < "job-board: limit: cargo jobs [M]"
		random < "job-board: weight: cargo industrial jobs [M]"
	source
		attributes hitech logistics
		not attributes isolated
	destination
		distance 1 2
			"all wormholes"
		attributes urban large
		not attributes isolated
		not attributes logistics
	on offer
		"job-board: cargo jobs [M]" ++
	on visit
		dialog phrase "generic cargo on visit"
	on complete
		payment 0 170
		dialog phrase "generic cargo delivery payment"
