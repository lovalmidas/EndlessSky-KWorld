### Naming ###
#	ID: "KW.Generic.Jobs.<priority><index><cargo type>.<short name>"
#	Example: "KW.Generic.Jobs.90f.Food Cargo [0]"
#		Sizes:
#			_ = special
#			0 = special
#			1 = 
#			2 = mega, deadline
#			3 = mega
#			4 = 
#			5 = bulk, deadline
#			6 = bulk
#			7 = 
#			8 = normal, deadline
#			9 = normal
#		Cargo Types:
#			_ = logistics
#			f = food
#			c = clothing
#			m = metal
#			p = plastic
#			q = equipment
#			i = industrial
#			d = medical
#			e = electronics
#			h = heavy metal
#			x = luxury goods
#			n = construction
#			y = military
#			s = ship alloys
#			r = research samples
#
# Game orders missions by internal name, so priority 0 is evaluated first
# Spread priority and index so other cargoes have a (near-equal) chance to appear

### LOGISTICS OPERATIONS ###
# Logistics move in bulk, but they also have to be moved in a timely manner.
#
#		[S] any (not logistics) -> logistics
#		[M] factory / logistics -> logistics
#		[L] factory / logistics + large -> logistics + large
#		exclude `isolated`; they take no generic jobs
#		exclude `logistics` destinations; logistics-to-logistics have their own jobs
#
# Use these amounts
#		Normal:	15 2 .07
#		Bulk:	40 2 .05
#		Mega:	150 3 .025
#
# Use these distances
#		[S]	1-4, 1-4, 3-6, 3-6
#		[M]	3-6, 7-9, 10-12
#		[L]	4-8, 9-12, 13-16
#		
# Payment
#		[S]	0 + 175 * d
#		[M]	0 + 200 * d
#		[L]	0 + 225 * d
#
# 
mission "KW.Generic.Jobs.80_.Logistics Transfer [0]"
	name "📦Shipment Order to <planet>"
	job
	repeat
	description "${KW.Phrase.Job: timed logistics cargo offer}"
	"offer precedence" 100
	color selected "KW.Color: job cargo selected"
	color unselected "KW.Color: job cargo unselected"
	color unavailable "KW.Color: job cargo unavailable"
	cargo random 15 2 .07
	deadline 2 1
	"distance calculation settings"
		"all wormholes"
	to offer
		has "job-board: enable generic cargo jobs"
		"job-board: cargo jobs [S]" < "job-board: limit: cargo jobs [S]"
		random < "job-board: weight: cargo logistics jobs [S]"
	source
		not attributes isolated
		not attributes logistics
	destination
		attributes logistics
		distance 1 4
			"all wormholes"
		not attributes isolated
	on offer
		"job-board: cargo jobs [S]" ++
	on visit
		dialog phrase "generic cargo on visit"
	on complete
		payment 0 175
		dialog phrase "generic cargo delivery payment"
