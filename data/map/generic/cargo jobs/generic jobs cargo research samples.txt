### Naming ###
#	ID: "KW.Generic.Jobs.<priority><index><cargo type>.<short name>"
#	Example: "KW.Generic.Jobs.90f.Food Cargo [0]"
#		Sizes:
#			_ = special
#			0 = special
#			1 = 
#			2 = mega, deadline
#			3 = mega
#			4 = 
#			5 = bulk, deadline
#			6 = bulk
#			7 = 
#			8 = normal, deadline
#			9 = normal
#		Cargo Types:
#			f = food
#			c = clothing
#			m = metal
#			p = plastic
#			q = equipment
#			i = industrial
#			d = medical
#			e = electronics
#			h = heavy metal
#			x = luxury goods
#			n = construction
#			y = military
#			s = ship alloys
#			r = research samples
#
# Game orders missions by internal name, so priority 0 is evaluated first
# Spread priority and index so other cargoes have a (near-equal) chance to appear

### COMMODITY: RESEARCH SAMPLES ###
# Most outposts are research stations. Over time they may develop other attributes. Research-related jobs will offer a bonus for your troubles
#
#		[all] research -> analysis
#		exclude `isolated`; they take no generic jobs
#
# Use these amounts
#		Normal:	15 3 .1
#
# Use these distances
#		[all] 1-2, 3-4, 5-7, 8-12
#		
# Payment
#		[all]	1500 + 160 * d
#
#
# These are timed missions.
mission "KW.Generic.Jobs.80r.Research Samples Cargo [0]"
	name "📦Delivery of Research Samples to <planet>"
	job
	repeat
	description "${KW.Phrase.Job: timed research sample offer}"
	"offer precedence" 100
	color selected "KW.Color: job cargo selected"
	color unselected "KW.Color: job cargo unselected"
	color unavailable "KW.Color: job cargo unavailable"
	cargo "research samples" 15 3 .1
	deadline 1 2
	"distance calculation settings"
		"all wormholes"
	to offer
		has "job-board: enable generic cargo jobs"
		"job-board: cargo jobs [S]" < "job-board: limit: cargo jobs [S]"
		random < "job-board: weight: cargo research sample jobs [S]"
	source
		attributes research
		not attributes isolated
	destination
		attributes analysis
		distance 1 2
			"all wormholes"
		not attributes isolated
	on offer
		"job-board: cargo jobs [S]" ++
	on visit
		dialog phrase "generic cargo on visit"
	on complete
		payment 1500 160
		dialog phrase "generic cargo delivery payment"
