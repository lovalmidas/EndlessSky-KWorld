### Naming ###
#	ID: "KW.Generic.Jobs.<priority><index><passenger type>.<short name>"
#	Example: "KW.Generic.Jobs.90c.Clothing Cargo [0]"
#		Sizes:
#			_ = special
#			0 = special
#			1 = 
#			2 = mega, deadline
#			3 = mega
#			4 = 
#			5 = bulk, deadline
#			6 = bulk
#			7 = 
#			8 = normal, deadline
#			9 = normal
#		Passenger Types:
#			Z = generic passengers
#			W = workers
#			F = farmers
#			T = tourists
#			S = students
#			R = researchers
#			SOS = rescue
#
# Game orders missions by internal name, so priority 0 is evaluated first
# Spread priority and index so other cargoes have a (near-equal) chance to appear

### PASSENGER: FARMERS ###
# Every factory needs workers to operate them
#	Farmers:
#			urban rural
#			farming synthetic
# Farmers carry farming equipment
#
#		[S] urban / rural -> farming / synthetic
#		[M] urban / rural -> farming / synthetic
#		[L] urban / rural + large -> farming / synthetic
#		exclude `isolated`; they take no generic jobs
#		exclude `Uninhabited` governments;
#
# Use these amounts
#		[S]	2 2 .7, 5 1 .25
#		[M]	2 10 .55, 10 4 .3
#		[L]	5 20 .3, 20 5 .2
#
# Use these distances
#		[S]	1-3, 4-5
#		[M]	1-3, 4-6
#		[L]	1-3, 4-6
#		
# Payment
#		[S]	0 + 155 * d
#		[M]	0 + 170 * d
#		[L]	0 + 185 * d
#
# 

mission "KW.Generic.Jobs.90F.Farmer Passengers [0]"
	name "👤Transport Farmers to <planet>"
	job
	repeat
	description "${KW.Phrase.Job: farmers offer}"
	"offer precedence" 200
	color selected "KW.Color: job passenger selected"
	color unselected "KW.Color: job passenger unselected"
	color unavailable "KW.Color: job passenger unavailable"
	passengers 2 2 .7
	cargo "farming equipment" 5 1 .25
	"distance calculation settings"
		"all wormholes"
	to offer
		has "job-board: enable generic passenger jobs"
		"job-board: passenger jobs [S]" < "job-board: limit: passenger jobs [S]"
		random < "job-board: weight: passenger farmer jobs [S]"
	source
		attributes urban rural
		not government "Uninhabited"
		not attributes isolated
	destination
		attributes farming synthetic
		distance 1 3
			"all wormholes"
		not government "Uninhabited"
		not attributes isolated
	on offer
		"job-board: passenger jobs [S]" ++
	on visit
		dialog phrase "generic passenger on visit"
	on complete
		payment 0 155
		dialog phrase "generic passenger dropoff payment"
